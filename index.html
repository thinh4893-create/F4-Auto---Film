document.getElementById('exportExcelBtn').addEventListener('click', () => { if (globalDataList.length === 0) { alert("Chưa có dữ liệu để xuất file Excel!"); return; } // 1. Chuyển đổi dữ liệu sang định dạng Excel let excelData = globalDataList.map((item, index) => { const totalPrice = Number(item.totalPrice) || 0; const workersArray = item.workerName ? item.workerName.split(',').map(w => w.trim()).filter(Boolean) : []; const workerCount = workersArray.length || 1; const perWorker = item.perWorker ? Number(item.perWorker) : Math.round(totalPrice / workerCount); return { "STT": index + 1, "Ngày Thi Công": formatDateOnly(item.timestamp), "Tên/Biển Số Xe": item.carName, "Loại Film": item.filmType, "Vị Trí Dán": item.position, "Tổng Tiền (VNĐ)": totalPrice, "Số Thợ": workerCount, "Thợ Thi Công": item.workerName, "Tiền Chia Mỗi Thợ (VNĐ)": perWorker, "Ghi Chú": item.note || '' }; }); // 2. Định nghĩa vị trí dòng dữ liệu (Dòng 1 là header, dữ liệu bắt đầu từ 2) const startRow = 2; const endRow = excelData.length + 1; // 3. Tạo dòng Tổng cộng với công thức SUM const totalRow = { "STT": "TỔNG CỘNG", "Ngày Thi Công": "", "Tên/Biển Số Xe": "", "Loại Film": "", "Vị Trí Dán": "", "Tổng Tiền (VNĐ)": { f: `SUM(F${startRow}:F${endRow})` }, // Cột F là Tổng Tiền "Số Thợ": "", "Thợ Thi Công": "", "Tiền Chia Mỗi Thợ (VNĐ)": { f: `SUM(I${startRow}:I${endRow})` }, // Cột I là Tiền Chia "Ghi Chú": "" }; excelData.push(totalRow); // 4. Xuất file const worksheet = XLSX.utils.json_to_sheet(excelData); const workbook = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(workbook, worksheet, "NhatKyF4Auto"); XLSX.writeFile(workbook, `F4Auto_BaoCao_${new Date().toISOString().slice(0,10)}.xlsx`); });
